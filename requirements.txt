# 创建命令（示例）：
#   conda create -n elve_me python=3.10 -y
#   conda run -n elve_me pip install torch==2.0.1 torchvision==0.15.2 \
#       --index-url https://download.pytorch.org/whl/cu118
#   conda run -n elve_me pip install xformers==0.0.20 \
#       --index-url https://download.pytorch.org/whl/cu118
#   conda run -n elve_me pip install -r requirements_unified.txt
#
# 注意：
# 1) torch/torchvision/xformers 必须从 cu118 index 装（见上），不在本文件 --index-url
# 2) ELVE 的 fatezero/controlnet 用本地 diffusersF/diffusersC（diffusers 0.11.1 fork），
#    通过 test_fatezero.py 的 setup_diffusers_path 切 sys.path，不依赖这里的 pip diffusers
# 3) MotionEditor 用 pip diffusers 0.15.1（method=MotionEditor 时不切 fork）

#    site-packages（basicsr 主要是纯 python）
